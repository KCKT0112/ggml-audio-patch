cmake_minimum_required(VERSION 3.14)
project(ggml_audio_patch_tests LANGUAGES C CXX)

set(GGML_SOURCE_DIR "" CACHE PATH "ggml v0.19.0 source with audio patches applied")
if(NOT EXISTS "${GGML_SOURCE_DIR}/include/ggml.h")
    message(FATAL_ERROR "Set GGML_SOURCE_DIR to a patched ggml v0.19.0 checkout")
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 17)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
add_subdirectory("${GGML_SOURCE_DIR}" ggml)

foreach(test_name test_learned_ops test_qvac_ops)
    add_executable(${test_name} ${test_name}.c)
    target_link_libraries(${test_name} PRIVATE ggml ggml-cpu)
    if(UNIX)
        target_link_libraries(${test_name} PRIVATE m)
    endif()
    if(GGML_METAL)
        target_compile_definitions(${test_name} PRIVATE USE_METAL)
        target_link_libraries(${test_name} PRIVATE ggml-metal)
    endif()
endforeach()
enable_testing()
add_test(NAME learned_cpu COMMAND test_learned_ops cpu)
add_test(NAME qvac_cpu COMMAND test_qvac_ops cpu)
if(GGML_METAL)
    add_test(NAME learned_metal COMMAND test_learned_ops metal)
    add_test(NAME qvac_metal COMMAND test_qvac_ops metal)
endif()
